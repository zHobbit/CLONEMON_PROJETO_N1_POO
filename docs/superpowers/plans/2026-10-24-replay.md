# Battle Replay Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a player re-watch any of their 20 most recent finished battles in-game, turn by turn, with play/pause, step, speed, history and a mirrored viewpoint.

**Architecture:** The backend records an event log per battle (start snapshot + each turn's events + player levels after the turn) in a new `battle_replay` table, written by `BattleService` in the battle's transaction, and serves it through `GET /api/replays[/{id}]`. The frontend extracts the arena (`BattleStage`) out of `BattleScene` so live battles and replays animate through the same code; a pure `battleAfter(replay, k)` rebuilds a `Battle`-shaped snapshot after `k` turns, which lets `BattleStage`, `BattleFx` and `turnCues` work unchanged.

**Tech Stack:** Java 21 + Spring Boot 4.1 + Spring Data JPA + Flyway + PostgreSQL 17 (jsonb) + Jackson 3 (`tools.jackson`) + JUnit 5/AssertJ/Testcontainers; TypeScript + Phaser 4 + Vite + Vitest + Playwright (Edge).

**Spec:** `docs/superpowers/specs/2026-10-24-replay-design.md`

- Replays are owner-only: other trainers get **404**. Active battles are never listed nor served.

```java
import br.clonemon.domain.Battle;

/** Replays de batalhas: retrato inicial dos times e os eventos de cada turno. */
public interface ReplayRepository {

    /** Monstro como estava no inicio da batalha (sem golpes). */
    record Combatant(long speciesId, String name, Element element, int level, int currentHp, int maxHp,
                     StatusCondition status) {
        public static Combatant of(Monster m) {
            return new Combatant(m.species().id(), m.species().name(), m.species().element(), m.level(),
                    m.currentHp(), m.maxHp(), m.status());
        }
    }

    /** Times completos no inicio. {@code npcName} e nulo contra selvagem. */
    record Snapshot(List<Combatant> playerTeam, List<Combatant> enemyTeam, int playerActive, int enemyActive,
                    String npcName) {
        public Snapshot {
            playerTeam = List.copyOf(playerTeam);
            enemyTeam = List.copyOf(enemyTeam);
        }

        /** Monstro que comecou a batalha. */
        public String lead() { return playerTeam.get(playerActive).name(); }
    }

    /** Nivel e HP maximo de um monstro do jogador depois do turno (muda ao subir de nivel). */
    record Growth(int level, int maxHp) {}

    record Turn(List<Battle.Event> events, List<Growth> playerTeam) {
        public Turn {
            events = List.copyOf(events);
            playerTeam = List.copyOf(playerTeam);
        }
    }

    record Summary(long id, Instant finishedAt, String opponent, String lead, Battle.Status result, int turnCount) {}

    record Replay(Summary summary, Snapshot snapshot, List<Turn> turns) {}

    void start(long battleId, long trainerId, String opponent, Snapshot snapshot);

    void appendTurn(long battleId, Turn turn);

    void finish(long battleId, Battle.Status result);

    /** Apaga os replays terminados do treinador alem dos {@code keep} mais novos. */
    void prune(long trainerId, int keep);

    /** Terminados, do mais novo para o mais antigo. */
    List<Summary> listFinished(long trainerId, int limit);

    Optional<Replay> findFinished(long trainerId, long battleId);
}
```

```java
    static final class Replays implements ReplayRepository {
        static final class Entry {
            final long battleId;
            final long trainerId;
            final String opponent;
            final Snapshot snapshot;
            final List<Turn> turns = new ArrayList<>();
            Battle.Status result;
            Instant finishedAt;

            Entry(long battleId, long trainerId, String opponent, Snapshot snapshot) {
                this.battleId = battleId;
                this.trainerId = trainerId;
                this.opponent = opponent;
                this.snapshot = snapshot;
            }

            Summary summary() {
                return new Summary(battleId, finishedAt, opponent, snapshot.lead(), result, turns.size());
            }
        }

        final Map<Long, Entry> entries = new LinkedHashMap<>();
        private Instant clock = Instant.parse("2026-01-01T12:00:00Z");

        @Override public void start(long battleId, long trainerId, String opponent, Snapshot snapshot) {
            entries.put(battleId, new Entry(battleId, trainerId, opponent, snapshot));
        }

        @Override public void appendTurn(long battleId, Turn turn) { entries.get(battleId).turns.add(turn); }

        @Override public void finish(long battleId, Battle.Status result) {
            Entry e = entries.get(battleId);
            e.result = result;
            clock = clock.plusSeconds(1);
            e.finishedAt = clock;
        }

        @Override public void prune(long trainerId, int keep) {
            finished(trainerId).stream().skip(keep).forEach(e -> entries.remove(e.battleId));
        }

        @Override public List<Summary> listFinished(long trainerId, int limit) {
            return finished(trainerId).stream().limit(limit).map(Entry::summary).toList();
        }

        @Override public Optional<Replay> findFinished(long trainerId, long battleId) {
            return Optional.ofNullable(entries.get(battleId))
                    .filter(e -> e.trainerId == trainerId && e.finishedAt != null)
                    .map(e -> new Replay(e.summary(), e.snapshot, List.copyOf(e.turns)));
        }

        private List<Entry> finished(long trainerId) {
            return entries.values().stream()
                    .filter(e -> e.trainerId == trainerId && e.finishedAt != null)
                    .sorted(Comparator.comparing((Entry e) -> e.finishedAt).reversed())
                    .toList();
        }
    }
```

```java
package br.clonemon.infrastructure.persistence;

import br.clonemon.application.NotFoundException;
import br.clonemon.application.port.ReplayRepository;
import br.clonemon.domain.Battle;
import org.springframework.data.domain.Limit;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.stereotype.Repository;
import org.springframework.transaction.annotation.Transactional;
import tools.jackson.core.type.TypeReference;
import tools.jackson.databind.json.JsonMapper;

import java.time.Instant;
import java.util.ArrayList;
import java.util.List;
import java.util.Optional;

@Repository
class JpaReplayRepository implements ReplayRepository {

    interface Jpa extends JpaRepository<ReplayEntity, Long> {
        List<ReplayEntity> findByTrainerIdAndFinishedAtNotNullOrderByFinishedAtDescBattleIdDesc(long trainerId, Limit limit);

        Optional<ReplayEntity> findByBattleIdAndTrainerIdAndFinishedAtNotNull(long battleId, long trainerId);
    }

    private static final TypeReference<List<Turn>> TURNS = new TypeReference<>() {};

    private final Jpa jpa;
    private final JsonMapper json;

    JpaReplayRepository(Jpa jpa, JsonMapper json) {
        this.jpa = jpa;
        this.json = json;
    }

    @Override
    public void start(long battleId, long trainerId, String opponent, Snapshot snapshot) {
        jpa.save(new ReplayEntity(battleId, trainerId, opponent, snapshot.lead(), json.writeValueAsString(snapshot)));
    }

    @Override
    @Transactional
    public void appendTurn(long battleId, Turn turn) {
        ReplayEntity e = entity(battleId);
        List<Turn> turns = new ArrayList<>(json.readValue(e.getTurns(), TURNS));
        turns.add(turn);
        e.setTurns(json.writeValueAsString(turns), turns.size());
        jpa.save(e);
    }

    @Override
    @Transactional
    public void finish(long battleId, Battle.Status result) {
        ReplayEntity e = entity(battleId);
        e.finish(result.name(), Instant.now());
        jpa.save(e);
    }

    @Override
    @Transactional
    public void prune(long trainerId, int keep) {
        List<ReplayEntity> finished = jpa.findByTrainerIdAndFinishedAtNotNullOrderByFinishedAtDescBattleIdDesc(trainerId, Limit.unlimited());
        if (finished.size() > keep) jpa.deleteAll(finished.subList(keep, finished.size()));
    }

    @Override
    public List<Summary> listFinished(long trainerId, int limit) {
        return jpa.findByTrainerIdAndFinishedAtNotNullOrderByFinishedAtDescBattleIdDesc(trainerId, Limit.of(limit))
                .stream().map(JpaReplayRepository::summary).toList();
    }

    @Override
    public Optional<Replay> findFinished(long trainerId, long battleId) {
        return jpa.findByBattleIdAndTrainerIdAndFinishedAtNotNull(battleId, trainerId)
                .map(e -> new Replay(summary(e), json.readValue(e.getSnapshot(), Snapshot.class), json.readValue(e.getTurns(), TURNS)));
    }

    private ReplayEntity entity(long battleId) {
        return jpa.findById(battleId).orElseThrow(() -> new NotFoundException("Replay not found: " + battleId));
    }

    private static Summary summary(ReplayEntity e) {
        return new Summary(e.getBattleId(), e.getFinishedAt(), e.getOpponent(), e.getLead(),
                Battle.Status.valueOf(e.getResult()), e.getTurnCount());
    }
}
```

```ts
import type { Battle, Combatant, Replay, ReplayMonster } from '../api/types';
import type { HistoryEntry } from '../battle/history';

function combatant(m: ReplayMonster): Combatant {
  return { monsterId: null, ...m, fainted: m.currentHp <= 0, moves: null };
}

/**
 * Estado da tela depois de `k` turnos (0 = inicio), no formato de uma batalha ao vivo, para a mesma arena
 * (BattleStage), os mesmos efeitos e os mesmos sons tocarem o replay.
 */
export function battleAfter(replay: Replay, k: number): Battle {
  const turns = Math.max(0, Math.min(k, replay.turns.length));
  const s = replay.snapshot;
  const player = s.playerTeam.map((m) => ({ ...m }));
  const enemy = s.enemyTeam.map((m) => ({ ...m }));
  let playerActive = s.playerActive;
  let enemyActive = s.enemyActive;

  for (const turn of replay.turns.slice(0, turns)) {
    for (const e of turn.events) {
      playerActive = e.playerActive;
      enemyActive = e.enemyActive ?? enemyActive;
      Object.assign(player[playerActive], { currentHp: e.playerHp, status: e.playerStatus });
      Object.assign(enemy[enemyActive], { currentHp: e.enemyHp, status: e.enemyStatus });
    }
    turn.playerTeam.forEach((g, i) => Object.assign(player[i], g));
  }

  const team = player.map(combatant);
  return {
    id: replay.id,
    status: turns === replay.turns.length ? replay.result : 'AWAITING_ACTION',
    playerActive,
    playerTeam: team,
    enemy: combatant(enemy[enemyActive]),
    log: [],
    npcId: s.npcName ? s.npcName.toLowerCase() : null,
    npcName: s.npcName,
    enemyTeamSize: enemy.length,
    enemyActive,
  };
}

/** Falas dos turnos ja mostrados, para o painel de historico. */
export function replayHistory(replay: Replay, k: number): HistoryEntry[] {
  return replay.turns.slice(0, Math.max(0, k)).flatMap((t, i) => t.events.map((e) => ({ turn: i + 1, text: e.text })));
}

/** Texto da caixa depois de um salto: o oponente no inicio, a ultima fala do turno depois dele. */
export function captionAt(replay: Replay, k: number): string {
  if (k <= 0) return `REPLAY: ${replay.opponent}`;
  return replay.turns[Math.min(k, replay.turns.length) - 1].events.at(-1)?.text ?? '';
}
```

```ts
function drawTeamBalls(g: Phaser.GameObjects.Graphics, battle: Battle, origin: { x: number; y: number }): void {
  g.clear();
  const size = battle.npcId ? (battle.enemyTeamSize ?? 0) : 0;

    const down = i < active || (i === active && battle.enemy.fainted);

  show(battle: Battle): void {
    this.battle = battle;
    this.after = null;
    for (const side of ['player', 'enemy'] as const) {
      const m = side === 'player' ? lead(battle) : battle.enemy;

      case 'attack': {
        // Investida na direcao do oponente: de perto sobe para a direita, de longe desce para a esquerda.
        const dir = SLOTS[this.slot(step.side)].lunge;

        return tween(this.scene, { targets: s, x: s.x + dir * LUNGE, y: s.y - (dir * LUNGE) / 2, duration: 90, yoyo: true, ease: 'Quad.Out' });

        return tween(this.scene, { targets: this.sprite(step.side), alpha: 0, duration: 70, yoyo: true, repeat: 2 });

      case 'status':

        return Promise.resolve();
      case 'stat': {
        const s = this.sprite(step.side);
        return tween(this.scene, { targets: s, y: s.y + (step.up ? -3 : 3), duration: 90, yoyo: true, repeat: 1 });

  }
}
```

```ts
  /** Exposto para os testes E2E. */

    super(SCENES.replays);

    this.summaries = null;
    addMenuBackground(this);
    addFrame(this, 0, 0, WIDTH, HEIGHT);
    addText(this, 8, 6, 'REPLAYS   (ESC VOLTA)', { color: CSS.accent });

    let list: ReplaySummary[];
    try {
      list = await api.replays();
    } catch {
      if (!this.scene.isActive()) return; // 401: ja fomos para o login
      status.setText('ERRO DE CONEXAO');
      await waitConfirm(this);
      fadeTo(this, SCENES.world);
      return;
    }
    this.summaries = list;
    if (list.length === 0) {
      status.setText('NENHUM REPLAY AINDA');
      await waitConfirm(this);
      fadeTo(this, SCENES.world);
      return;
    }
    status.destroy();
    this.choose(list);
  }

  private choose(list: ReplaySummary[]): void {
    const rows = addText(this, 14, 22, '', { lineSpacing: ROW_HEIGHT - 8 });
    const cursor = addText(this, 4, 22, '>');
    let index = 0;
    let start = 0;
    const render = () => {
      start = listStart(index, list.length, ROWS, start);
      rows.setText(list.slice(start, start + ROWS).map(listLabel).join('\n'));
      cursor.setY(22 + (index - start) * ROW_HEIGHT);
    };
    render();
    const off = onKey(this, (k) => {
      if (k === 'up' || k === 'down') {
        const next = Math.max(0, Math.min(list.length - 1, index + (k === 'up' ? -1 : 1)));
        if (next !== index) audio.sfx('cursor');
        index = next;
        render();
      } else if (k === 'confirm') {
        off();
        audio.sfx('confirm');
        fadeTo(this, SCENES.replay, { id: list[index].id });
      } else if (k === 'cancel') {
        off();
        audio.sfx('cancel');
        fadeTo(this, SCENES.world);
      }
    });
  }
```

- [ ] **Step 4: Player scene** (`ReplayScene.ts`)

```ts
import Phaser from 'phaser';
import { ApiError } from '../api/client';
import type { Replay } from '../api/types';
import { planTurn, viewOf } from '../battle/playback';
import { CSS, SCENES } from '../config';
import { fadeTo } from '../fx/transitions';
import { SPEEDS, footerLabel, nextSpeed } from '../replay/controls';
import { battleAfter, captionAt, replayHistory } from '../replay/timeline';
import { api } from '../services';
import { showHistory } from '../ui/HistoryPanel';
import { onKey, waitConfirm } from '../ui/input';
import type { Key } from '../ui/navigation';
import { TextBox } from '../ui/TextBox';
import { addText, delay } from '../ui/widgets';
import { BattleStage, type Perspective } from './BattleStage';

interface ReplayData {
  id: number;
}

/** Tempo que cada fala fica na tela antes da proxima (dividido pela velocidade). */
const TEXT_HOLD_MS = 700;
/** Velocidade usada para terminar depressa o turno em andamento (seta para a direita). */
const SKIP_SPEED = 20;

/**
 * Toca uma batalha gravada com a mesma arena da batalha ao vivo. Enter pausa, setas pulam turnos
 * (direita/esquerda) e mudam a velocidade (cima/baixo), V espelha, H abre o historico e Esc volta a lista.
 */
export class ReplayScene extends Phaser.Scene {
  /** Exposto para os testes E2E. */
  replay: Replay | null = null;
  turn = 0;
  playing = true;
  perspective: Perspective = 'player';
  private speed = 0;
  private stage!: BattleStage;
  private textBox!: TextBox;
  private footer!: Phaser.GameObjects.Text;
  private animating = false;
  private modal = false;
  private skipping = false;
  private pendingJump: number | null = null;
  private pendingFlip = false;
  private wake: (() => void) | null = null;

  constructor() {
    super(SCENES.replay);
  }

  create(data: ReplayData): void {
    this.replay = null;
    this.turn = 0;
    this.playing = true;
    this.perspective = 'player';
    this.speed = 0;
    this.animating = this.modal = this.skipping = this.pendingFlip = false;
    this.pendingJump = null;
    this.wake = null;
    this.setSpeed(1);
    void this.load(data.id);
  }

  private async load(id: number): Promise<void> {
    let replay: Replay;
    try {
      replay = await api.replay(id);
    } catch (e) {
      if (!this.scene.isActive()) return;
      addText(this, 8, 8, e instanceof ApiError && e.status === 404 ? 'REPLAY NAO ENCONTRADO' : 'ERRO DE CONEXAO', { color: CSS.paper });
      await waitConfirm(this);
      fadeTo(this, SCENES.replays);
      return;
    }
    this.replay = replay;
    this.stage = new BattleStage(this, battleAfter(replay, 0));
    this.textBox = new TextBox(this);
    this.textBox.setText(captionAt(replay, 0));
    this.footer = addText(this, 236, 1, '', { color: CSS.paper, stroke: CSS.ink, strokeThickness: 2 }).setOrigin(1, 0);
    onKey(this, (k) => this.onKey(k));
    void this.loop();
  }

  private setSpeed(factor: number): void {
    this.tweens.timeScale = factor;
    this.time.timeScale = factor;
  }

  private renderFooter(): void {
    if (this.replay) this.footer.setText(footerLabel(this.turn, this.replay.turns.length, this.speed, this.playing));
  }

  private async loop(): Promise<void> {
    const replay = this.replay!;
    while (this.scene.isActive()) {
      this.applyPending();
      this.renderFooter();
      if (this.playing && this.turn < replay.turns.length) {
        await this.playTurn(this.turn);
        this.turn++;
        if (this.turn >= replay.turns.length) this.playing = false;
        continue;
      }
      await new Promise<void>((resolve) => (this.wake = resolve));
      this.wake = null;
    }
  }

  private async playTurn(k: number): Promise<void> {
    const replay = this.replay!;
    const before = battleAfter(replay, k);
    const after = battleAfter(replay, k + 1);
    const events = replay.turns[k].events;
    this.animating = true;
    this.stage.beginTurn(before, after, events);
    for (const step of planTurn(viewOf(before), events)) {
      if (step.kind === 'text') await this.textBox.say(step.text);
      else if (step.kind === 'wait') await delay(this, TEXT_HOLD_MS);
      else await this.stage.play(step);
    }
    this.stage.show(after);
    this.animating = false;
    this.skipping = false;
    this.setSpeed(SPEEDS[this.speed]);
  }

  /** Saltos e espelho pedidos durante um turno so valem quando ele termina. */
  private applyPending(): void {
    const replay = this.replay!;
    if (this.pendingFlip) {
      this.pendingFlip = false;
      this.perspective = this.perspective === 'player' ? 'enemy' : 'player';
      this.stage.setPerspective(this.perspective);
    }
    if (this.pendingJump !== null) {
      this.turn = this.pendingJump;
      this.pendingJump = null;
      this.stage.show(battleAfter(replay, this.turn));
      this.textBox.setText(captionAt(replay, this.turn));
    }
  }

  private onKey(k: Key): void {
    const replay = this.replay;
    if (!replay || this.modal) return;
    const total = replay.turns.length;
    switch (k) {
      case 'confirm':
        this.playing = !this.playing;
        if (this.playing && this.turn >= total) this.pendingJump = 0; // no fim, Enter recomeca
        break;
      case 'right':
        if (this.animating) {
          this.skipping = true;
          this.setSpeed(SKIP_SPEED);
        } else {
          this.playing = false;
          this.pendingJump = Math.min(total, (this.pendingJump ?? this.turn) + 1);
        }
        break;
      case 'left':
        this.playing = false;
        if (this.animating) {
          this.skipping = true;
          this.setSpeed(SKIP_SPEED);
          this.pendingJump = this.turn;
        } else {
          this.pendingJump = Math.max(0, (this.pendingJump ?? this.turn) - 1);
        }
        break;
      case 'up':
      case 'down':
        this.speed = nextSpeed(this.speed, k === 'up' ? 1 : -1);
        if (!this.skipping) this.setSpeed(SPEEDS[this.speed]);
        break;
      case 'view':
        this.pendingFlip = true;
        break;
      case 'history':
        if (this.animating) return;
        this.playing = false;
        this.modal = true;
        void showHistory(this, replayHistory(replay, this.turn)).then(() => {
          this.modal = false;
          this.renderFooter();
        });
        break;
      case 'cancel':
        this.modal = true;
        fadeTo(this, SCENES.replays);
        return;
    }
    this.renderFooter();
    this.wake?.();
  }
}
```
